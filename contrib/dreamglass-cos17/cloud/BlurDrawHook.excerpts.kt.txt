    private fun seedlingRuntimeAllowed(): Boolean {
        val allowed = runCatching {
            val activeApi = api ?: return@runCatching false
            if (!Prefs.memberAllowed(activeApi)) return@runCatching false
            Prefs.read(activeApi).getBoolean(Prefs.KEY_SEEDLING_CARD_GLASS_ENABLE, false)
        }.getOrDefault(false)
        if (allowed) seedlingRuntimeWasAllowed = true
        else if (seedlingRuntimeWasAllowed) {
            seedlingRuntimeWasAllowed = false
            val host = cardBackgroundActiveHost?.get()
            cardBackgroundActiveHost = null // Existing bounded worker exits its next active check.
            for ((id, entry) in screenRegionMap) {
                if (entry.hostViewName.startsWith("seedling-")) {
                    screenRegionMap.remove(id, entry);appliedUniformsMap.remove(id)
                }
            }
            host?.postInvalidate() // Re-record native background instead of retaining old glass.
        }
        return allowed
    }

    private fun renderSeedlingCardGlass(canvas: Canvas, container: View) {
        // Revocation removes expanded glass, while the collapsed pill keeps the
        // native black rule below. It must never become a transparent leftover.
        seedlingRuntimeAllowed()
        // [spec/85 省电模式] 省电开启 → 不注入胶囊玻璃（直接让系统原样绘制）
        if (isPowerSaveGlassOff()) return
        val containerView = container as? android.view.ViewGroup ?: return
        val containerHeight = containerView.height
        if (containerHeight <= 0) {
            if (moduleLogEnabled) Lg.i(TAG, "seedling-card: skip container height=$containerHeight ${container.javaClass.name}")
            return
        }
        if (moduleLogEnabled) {
            Lg.i(TAG, "seedling-card: dispatch container=${container.javaClass.name} " +
                "w=${containerView.width} h=$containerHeight childCount=${containerView.childCount}")
        }
        // Collapsed CapsuleView always uses black; expanded cards have a separate renderer.
        // [2026-09-16] 目标 = **CapsuleView（胶囊本体）**，不是 CapsuleContainer（外框）：
        // `CapsuleContainer.onMeasure`(:445-449) 宽度 = 子宽 + padding，`CapsuleContainer.q()`(:506-516)
        // 把 CapsuleView **居中**摆放 —— 拿外框当本体玻璃必然画大、超出胶囊（用户反馈「玻璃比岛大」）。
        val capsules = ArrayList<View>(4)
        collectCapsuleViews(containerView, capsules)
        if (capsules.isEmpty()) {
            // [2026-09-16 用户需求] 胶囊 ↔ 展开大卡片互转期间插件会把 CapsuleView 从容器移除
            // （真机实证），此时**玻璃不显示**——用户明确要求「动画期间干脆不显示」，不做补绘。
            lastIslandW = -1
            lastIslandH = -1
            return
        }
        val containerLoc = IntArray(2)
        try {
            containerView.getLocationOnScreen(containerLoc)
        } catch (t: Throwable) {
            return
        }
        for (capsule in capsules) {
            // 不可见的不画（插件动画期间会短暂置 INVISIBLE）。
            if (capsule.visibility != View.VISIBLE) {
                logThrottled("seedling-invis", Log.WARN, 500L) {
                    "seedling-card: skip invisible CapsuleView vis=${capsule.visibility}"
                }
                continue
            }
            // CapsuleView 由 setLeftTopRightBottom 摆放，getWidth() 偶为 0（布局未完成）→ 退回 measured
            val cvW = if (capsule.width > 0) capsule.width else capsule.measuredWidth
            val cvH = if (capsule.height > 0) capsule.height else capsule.measuredHeight
            if (cvW <= 0 || cvH <= 0) {
                logThrottled("seedling-zero", Log.WARN, 500L) {
                    "seedling-card: ABORT zero size cv=${capsule.width}x${capsule.height} " +
                        "measured=${capsule.measuredWidth}x${capsule.measuredHeight} vis=${capsule.visibility}"
                }
                continue
            }
            // ★ 可见胶囊宽度 = `CapsuleView.capsuleDrawableWidth`：背景/前景 Drawable 的 bounds 被限制在
            //   这个宽度内并**水平居中**（CapsuleView.setCapsuleDrawableWidth:249-255），
            //   高度取 CapsuleView 全高（bounds 用的就是 getMeasuredHeight()）。
            //   该字段由动画逐帧更新 → 玻璃尺寸天然跟随小岛动效。
            // 可见宽度直接用 CapsuleView 自身宽度。**不要读 Drawable bounds**：
            // 真机实测 background 被本函数置空后读不到，foreground 的 bounds 是 3x3（无关小图），
            // 拿它当宽度会让玻璃宽度在 402 与 3 之间跳 → 动画期间玻璃几乎不可见（实测）。
            // 402 为用户确认「平齐」的尺寸。
            val drawW = cvW
            // ★ **不能用 CapsuleView.getLocationOnScreen()**：真机实测返回 (720,80) = 屏幕中心
            //   （玻璃画在岛中心，用户反馈"岛偏了"）。原因在插件里——`CapsuleContainer.q()`
            //   计算居中位置时 CapsuleView 尚未测量（measuredWidth=0 → (1440-0)/2=720），
            //   之后测量完成但位置**不再更新** → 该值陈旧。
            //   改为按容器几何直接算居中：系统布局中胶囊水平/垂直居中于容器
            //   （CapsuleContainer 自身也居中于 CapsulePluginContainer，真机 432@504 = (1440-432)/2）。
            val left = containerLoc[0] + (containerView.width - drawW) / 2f
            val top = containerLoc[1] + (containerHeight - cvH) / 2f
            val region = RectF(left, top, left + drawW, top + cvH)
            if (region.width() <= 0f || region.height() <= 0f) {
                logThrottled("seedling-badregion", Log.WARN, 500L) {
                    "seedling-card: ABORT bad region=$region draw=${drawW}x$cvH"
                }
                continue
            }
            // Match the capsule's actual native pill bounds, leaving its children untouched.
            val black = capsule.background as? android.graphics.drawable.GradientDrawable
            if(black!=null) black.setColor(android.graphics.Color.BLACK)
            else capsule.background=android.graphics.drawable.GradientDrawable().apply {
                setColor(android.graphics.Color.BLACK);cornerRadius=cvH/2f
            }
            screenRegionMap.remove(System.identityHashCode(capsule))
            RuntimeDiagnostics.mark("seedling-collapsed","收起胶囊强制黑底；展开卡片继续使用独立玻璃路径")
        }
    }

    private fun drawSeedlingGlassOnCanvas(
        canvas: Canvas,
        host: View,
        id: Int,
        w: Int,
        h: Int,
        screenRegion: RectF,
        offsetX: Float = host.left.toFloat(),
        offsetY: Float = host.top.toFloat(),
    ):Boolean {
        if (host.javaClass.name == CLASS_CARD_BACKGROUND_VIEW && !seedlingRuntimeAllowed()) return false
        // [spec/85 省电模式] 省电开启 → 不画 canvas 玻璃（覆盖 CardBackgroundView / Simple Banner 等
        // 所有走本函数的注入点）
        if (isPowerSaveGlassOff()) return false
        if (w <= 0 || h <= 0) return false
        GlassRegistry.bind(id, host)
        val entry = screenRegionMap[id] ?: run {
            logThrottled("seedling-nomap", Log.WARN, 500L) { "seedling-card: ABORT no region map id=$id" }
            return false
        }
        val renderSource = resolveRenderSource(id, screenRegion) ?: run {
            logThrottled("seedling-nosrc", Log.WARN, 500L) {
                "seedling-card: ABORT no render source id=$id region=$screenRegion"
            }
            return false
        }
        var srcRect = renderSource.srcRect
        var source = renderSource.source
        val requiredEpoch = when {
            host.javaClass.name == CLASS_CARD_BACKGROUND_VIEW -> cloudRequiredCaptureEpoch
            entry.hostViewName.startsWith("simple-banner:") -> bannerRequiredCaptureEpoch
            else -> Long.MIN_VALUE
        }
        if (!isKeyguardLockedNow() && captureMode() != Prefs.BG_CAPTURE_MODE_STATIC &&
            !overlayFrameEpochs.accepts(source, requiredEpoch)) return false
        val viewport = RectF(0f, 0f, w.toFloat(), h.toFloat())
        val cornerRadius = GlassRegistry.cornerOverrideFor(id)?.coerceAtMost(minOf(w, h) / 2f)
            ?: entry.cornerRadius.takeIf { it > 0f }
            ?: (Math.min(w, h).toFloat() * seedlingCardCornerRatio)
        var params = buildParams(w.toFloat(), h.toFloat(), cornerRadius, entry.cornerIsConic, entry.cornerWeight, elementId = id)
        // 音量元素（hostViewName="volume-*"：音量条/···按钮）弱化折射与色散：小面积元素强折射会把
        // 快照内容放大成扭曲倒影（验收反馈的黑渐变/棕色带/红色倒影），预设未接管该键时才衰减
        if (entry.hostViewName.startsWith("volume-")) {
            val ov = GlassRegistry.overridesFor(id)
            if (ov?.get(Prefs.KEY_REFRACTION_AMOUNT) == null) {
                params = params.copy(refractionAmount = params.refractionAmount * VOLUME_REFRACTION_DAMP)
            }
            if (ov?.get(Prefs.KEY_DISPERSION) == null) {
                params = params.copy(dispersion = params.dispersion * VOLUME_DISPERSION_DAMP)
            }
        }
        // Use the same bounded GPU backdrop binding as shade cards. This path used
        // to bypass it entirely after capture became demand-driven, leaving sparse
        // raw sampling and the ROM's heavy dark blur as the apparent material.
        val binding = bindQualityBackdrop(id, source, renderSource.screenRect, srcRect,
            renderSource.isSnapshotSource, params)
        source=binding.bitmap;srcRect=binding.sourceRect;params=binding.params
        val shader = obtainShader(id, metaball = entry.metaBall != null)
        val sourceShader = LiquidGlassShader.createSourceBitmapShader(source)
        // σ 分档三档源：命中绑档位 shader + 折算 rect，miss 回退现状源（本函数每帧全量重绑，无轻量分支）
        val seedlingWantDiv = lowResTierOf(0.807f * params.blurRadius)
        val seedlingTier = if (params.sourcePreblurred) null else buildLowResTierShader(source, srcRect, seedlingWantDiv)
        LiquidGlassShader.setUniforms(
            shader, params, sourceShader, seedlingTier?.shader ?: sourceShader,
            sourceRect = srcRect,
            lowResSourceRect = seedlingTier?.lowResSrcRect ?: srcRect,
            debugCoord = 0f,
            maskRect = null,
            maskCornerRadius = 0f,
            cornerConicBlend = 1f,
            maskOffsetX = 0f,
            maskOffsetY = 0f,
            maskAlpha = 0f,  // 桌面流体云无 shade 遮罩（遮罩仅作用于 SystemUI 下拉）
        )
        appliedUniformsMap[id] = AppliedKey(
            frame = 0L, sourceBitmapId = System.identityHashCode(source),
            viewport = viewport, srcRect = srcRect, maskRect = null,
            cornerIsConic = entry.cornerIsConic, cornerWeight = entry.cornerWeight,
            cornerRadius = cornerRadius, cornerConicBlend = 1f, maskAlpha = 0f,
            lowResDiv = if (seedlingTier != null) seedlingWantDiv else 4,
        )
        canvas.save()
        var drawn=false
        try {
            canvas.translate(offsetX, offsetY)
            seedlingGlassPaint.shader = shader
            val path = android.graphics.Path().apply {
                val r = cornerRadius
                addRoundRect(RectF(0f, 0f, w.toFloat(), h.toFloat()), r, r, android.graphics.Path.Direction.CW)
            }
            canvas.drawPath(path, seedlingGlassPaint)
            drawn=true
        } catch (t: Throwable) {
            Lg.e(TAG, "seedling-card: draw glass on canvas failed id=$id", t)
        } finally {
            // 清理 paint.shader（防止残留到下一帧其他绘制）；canvas.restore 还原 translate
            seedlingGlassPaint.shader = null
            canvas.restore()
        }
        return drawn
    }

    private fun mountCardBackgroundGlass(api: XposedInterface, classLoader: ClassLoader) {
        if (!seedlingCardGlassEnabled || cardBgGlassMounted) return
        try {
            val setter = View::class.java.getDeclaredMethod("setBackgroundDrawable", Drawable::class.java)
            api.hook(setter).setExceptionMode(ExceptionMode.PROTECTIVE).intercept { chain ->
                val host = chain.getThisObject() as? View
                val background = chain.getArg(0) as? Drawable
                if (host?.javaClass?.name == CLASS_CARD_BACKGROUND_VIEW && background != null &&
                    background !is ExpandedCloudGlassDrawable && seedlingRuntimeAllowed()) {
                    val replacement = runCatching { expandedCloudBackground(host, background) }.getOrNull()
                    if (replacement != null) {
                        val args = chain.getArgs().toTypedArray()
                        args[0] = replacement
                        chain.proceed(args)
                    } else chain.proceed()
                } else chain.proceed()
            }
            // Native CardBackgroundView.draw updates its animation bounds before calling
            // super.draw. This nonempty entry also repairs an inlined/replaced setter.
            val draw = View::class.java.getDeclaredMethod("draw", Canvas::class.java)
            api.hook(draw).setExceptionMode(ExceptionMode.PROTECTIVE).intercept { chain ->
                val host = chain.getThisObject() as? View
                if (host?.javaClass?.name == CLASS_CARD_BACKGROUND_VIEW && seedlingRuntimeAllowed()) {
                    runCatching {
                        val original = host.background
                        if (original !is ExpandedCloudGlassDrawable) {
                            host.background = expandedCloudBackground(host, original)
                        }
                        cardBackgroundActiveHost = WeakReference(host)
                        if (!cardBackgroundContinuousRunning) startCardBackgroundContinuousCapture()
                    }.onFailure { Lg.e(TAG, "seedling-cardview: background attachment failed", it) }
                }
                chain.proceed()
            }
            runCatching { api.deoptimize(View::class.java.getDeclaredMethod("setBackground", Drawable::class.java)) }
            cardBgGlassMounted = true
            Lg.i(TAG, "Seedling card glass mounted at native background paint")
        } catch (error: Throwable) {
            Lg.e(TAG, "Seedling card background hook mount failed", error)
        }
    }

    private fun renderCardBackgroundGlass(canvas: Canvas, bgView: View): Boolean {
        if (!seedlingRuntimeAllowed()) return false
        val w = bgView.width
        val h = bgView.height
        if (w <= 0 || h <= 0 || bgView.visibility != View.VISIBLE) return false
        val loc = IntArray(2)
        try {
            bgView.getLocationOnScreen(loc)
        } catch (t: Throwable) {
            return false
        }
        val screenRegion = RectF(
            loc[0].toFloat(), loc[1].toFloat(),
            (loc[0] + w).toFloat(), (loc[1] + h).toFloat(),
        )
        val id = System.identityHashCode(bgView)
        // 圆角：一次解析并缓存（避免每帧 getIdentifier）；SystemUIPlugin.apk 资源，包 com.oplus.systemui.plugins
        if (roundCornerFluidCloudPx <= 0f) {
            roundCornerFluidCloudPx = try {
                val resId = bgView.resources.getIdentifier(
                    "round_corner_radius_fluid_cloud", "dimen", "com.oplus.systemui.plugins",
                )
                if (resId != 0) {
                    bgView.resources.getDimensionPixelSize(resId).toFloat()
                } else {
                    20f * bgView.resources.displayMetrics.density
                }
            } catch (t: Throwable) {
                20f * bgView.resources.displayMetrics.density
            }
        }
        registerSeedlingCardRegion(id, screenRegion, Rect(0, 0, w, h), bgView, "seedling-cardview", roundCornerFluidCloudPx)
        return drawSeedlingGlassOnCanvas(canvas, bgView, id, w, h, screenRegion, offsetX = 0f, offsetY = 0f)
    }

    private fun startVisibleOverlayCapture(banner: Boolean) {
        if (captureMode() == Prefs.BG_CAPTURE_MODE_STATIC || !captureDemandActive() || isKeyguardLockedNow()) return
        val host = (if (banner) simpleBannerActiveHost else cardBackgroundActiveHost)?.get() ?: return
        if (!(if (banner) isSimpleBannerHostActive() else isCardBackgroundHostActive())) return
        val session = if (banner) bannerCaptureSession else cardCaptureSession
        val token = session.begin(host) ?: return
        if (banner) simpleBannerContinuousRunning = true else cardBackgroundContinuousRunning = true
        // Require a fresh scene for this overlay, without blanking other visible glass.
        // Publication is atomic from the renderer's perspective: only a frame stamped
        // after this reveal can replace its native fallback. Never recycle a previous app.
        val revealEpoch = ++captureEpoch
        noteCaptureReveal()
        if (banner) bannerRequiredCaptureEpoch = revealEpoch else cloudRequiredCaptureEpoch = revealEpoch
        lastSnapshotFingerprint=0;lastRegionSnapshotFingerprint=0
        triggerBackgroundCapture(contentChange=true)
        try {
            safeThread {
                try {
                    while (session.current(token) && captureMode() != Prefs.BG_CAPTURE_MODE_STATIC &&
                        captureDemandActive() && Prefs.glassEnabled(api) && !isPowerSaveGlassOff()) {
                        val current=(if (banner) simpleBannerActiveHost else cardBackgroundActiveHost)?.get()
                        if (current !== host || !(if (banner) isSimpleBannerHostActive() else isCardBackgroundHostActive()) ||
                            isKeyguardLockedNow()) break
                        scheduleElementCaptures()
                        Thread.sleep(continuousCaptureIntervalMs())
                    }
                } catch (_:InterruptedException) {
                    Thread.currentThread().interrupt()
                } finally {
                    if (session.finish(token)) {
                        if (banner) simpleBannerContinuousRunning=false else cardBackgroundContinuousRunning=false
                        // A replacement can have arrived without another native draw.
                        val next=(if (banner) simpleBannerActiveHost else cardBackgroundActiveHost)?.get()
                        if (next !== host && visibleCaptureHost(next)) next?.postInvalidateOnAnimation()
                    }
                }
            }.start()
        } catch (error:Throwable) {
            session.finish(token)
            if (banner) simpleBannerContinuousRunning=false else cardBackgroundContinuousRunning=false

        }
    }

                // 见此 SNAPSHOT_ID 即不重复入队（避免双路 captureDisplay）
                captureInFlightIds.add(id)
                try {
                    // All event/timer channels share this admission limit, so duplicate event
                    // bursts cannot bypass the change-mode budget. No pixels are downsampled here.
                    if (awaitCaptureSlot()) {
                        captureSnapshotImmediate("worker", regionMode = id == REGION_SNAPSHOT_ID)
                    }
                } finally {
                    captureInFlightIds.remove(id)
                }
            }
        } catch (t: Throwable) {
            captureRetryAtMs = SystemClock.uptimeMillis() + RuntimeWorkPolicy.failureDelayMs(t)
            synchronized(this) { elementCaptureQueue.clear(); pendingCaptureIds.clear() }
            if (t is OutOfMemoryError && buttonBackdropWorkerDelegate.isInitialized()) buttonBackdropWorker.clear()
            Lg.e(TAG, "bg-element: worker error (backoff enabled)", t)
        } finally {
            synchronized(this) {
                if (elementCaptureWorkerRunning > 0) elementCaptureWorkerRunning--
                if (elementCaptureQueue.isEmpty()) {
                    // 队列空（正常消费完）→ 清任何残留 pending（防御：极端时序下 poll 出队后、pending.remove
                    // 前有并发入队又被本 worker 消费，或竞态窗口入队但未入队成功的残留；保证下次入队
                    // pending.add 成功）。正常路径 pending 应为空，此分支仅日志不误删在途任务。
                    if (pendingCaptureIds.isNotEmpty()) {
                        Lg.w(TAG, "bg-element: worker exit, clear orphan pending=$pendingCaptureIds")
                        pendingCaptureIds.clear()
                    }
                    // [spec/59 追加 2026-08-14 诊断] worker 正常退出（队列空）——对比 worker-spawn 看 worker
                    // 存活时长；「短命 worker」= 消费完即退出，靠下次入队重新 kick（正常）；「长期无 exit」=
                    // worker 存活/卡住（与 captureDisplay-start 间隔对照判断是否卡 Binder 或 sfc 重试循环）
                    diagWorkerLog("worker-exit", " queue-empty running=$elementCaptureWorkerRunning")
                } else {
                    // 队列仍有未消费任务（poll==null 到 running-- 之间竞态窗口入队 + kick 早退残留）
                    // → 自续跑新 worker 消费，杜绝「孤儿队列 + pending 残留 + running=0 永久不消费」
                    elementCaptureWorkerRunning++
                    diagWorkerLog("worker-respawn", " queue=${elementCaptureQueue.size} running=$elementCaptureWorkerRunning")
                    try {
                        captureExecutor.execute {
                    safeModuleTask { android.os.Process.setThreadPriority(android.os.Process.THREAD_PRIORITY_BACKGROUND) }
                    safeModuleTask { runElementCaptureWorker() }
                }
                    } catch (t: Throwable) {
                        elementCaptureWorkerRunning--
                        Lg.e(TAG, "bg-element: respawn worker failed", t)
                    }
                }
            }
        }
    }
